cmake_minimum_required(VERSION 3.18)
project(notakillite_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT WHISPER_SOURCE)
  message(FATAL_ERROR "Set WHISPER_SOURCE to the pinned whisper.cpp checkout")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(WHISPER_BUILD_EXAMPLES ON CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${WHISPER_SOURCE} whisper-build)
add_executable(notakillite-worker main.cpp)
target_include_directories(notakillite-worker PRIVATE ${WHISPER_SOURCE}/examples)
target_link_libraries(notakillite-worker PRIVATE whisper)

if(WIN32)
  target_link_libraries(notakillite-worker PRIVATE psapi)
endif()
