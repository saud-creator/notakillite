cmake_minimum_required(VERSION 3.18)
project(notakillite_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(NOTAKILLITE_DEVICE_POLICY_TEST_ONLY "Configure only the standalone GPU selection test" OFF)
enable_testing()
add_executable(notakillite-device-policy-test EXCLUDE_FROM_ALL device-policy.test.cpp)
target_include_directories(notakillite-device-policy-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME notakillite-device-policy COMMAND notakillite-device-policy-test)

if(NOTAKILLITE_DEVICE_POLICY_TEST_ONLY)
  return()
endif()
if(NOT WHISPER_SOURCE)
  message(FATAL_ERROR "Set WHISPER_SOURCE to the pinned whisper.cpp checkout")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(WHISPER_BUILD_EXAMPLES ON CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${WHISPER_SOURCE} whisper-build)
if(WIN32 AND MSVC AND TARGET whisper-cli)
  target_link_options(whisper-cli PRIVATE
    "/MANIFEST:EMBED"
    "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/windows-utf8.manifest")
endif()
add_executable(notakillite-worker main.cpp)
target_include_directories(notakillite-worker PRIVATE ${WHISPER_SOURCE}/examples)
target_link_libraries(notakillite-worker PRIVATE whisper)

if(WIN32)
  target_link_libraries(notakillite-worker PRIVATE psapi)
endif()
